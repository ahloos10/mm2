# Power Automate Desktop Flow: Iterate All Policy Terms and Extract Details
#
# INPUT VARIABLES:
#   PolicyNumber   - Text
#   PCBaseURL      - Text
#
# OUTPUT VARIABLES:
#   TermsJSON      - Text (JSON array of all terms with key details)
#
# DESCRIPTION:
#   Navigates to a policy, reads the term history list, and for each term
#   captures: term number, effective date, expiration date, premium, status,
#   and transaction type. Returns all terms as a JSON array.

# -- STEP 1: Launch and navigate to policy --
WebBrowser.Launch BrowserType: WebBrowser.BrowserType.Chrome Url: $'''%PCBaseURL%''' WindowState: WebBrowser.WindowState.Maximized ClearCache: False ClearCookies: False WaitForPageToLoad: True PopupsHandling: WebBrowser.PopupWindowHandling.DoNotHandle BrowserInstance=> BrowserInstance

WebBrowser.GoToWebPage Browser: BrowserInstance Url: $'''%PCBaseURL%/PolicyFile.do''' WaitForPageToLoad: True
WebAutomation.PopulateTextField BrowserInstance: BrowserInstance Control: appmask['PolicySearch_PolicyNumber'] Text: PolicyNumber SimulateTyping: False UnfocusAfterPopulate: False WaitForPageToLoad: False
WebAutomation.Click BrowserInstance: BrowserInstance Control: appmask['PolicySearch_SearchButton'] WaitForPageToLoad: True
WebAutomation.Click BrowserInstance: BrowserInstance Control: appmask['PolicySearch_FirstResult'] WaitForPageToLoad: True

# -- STEP 2: Navigate to Policy Info or History tab (contains term list) --
WebAutomation.Click BrowserInstance: BrowserInstance Control: appmask['Policy_HistoryTab'] WaitForPageToLoad: True

# -- STEP 3: Extract the terms table via JavaScript --
# PolicyCenter renders terms as table rows. JavaScript is the most reliable
# way to extract a structured list without recording each row individually.
WebAutomation.ExecuteJavascript BrowserInstance: BrowserInstance Javascript: $'''
    var rows = document.querySelectorAll('[id*="PolicyTerms"] tr, [class*="term"] tr, table tr');
    var terms = [];
    rows.forEach(function(row) {
        var cells = row.querySelectorAll('td');
        if (cells.length >= 3) {
            terms.push({
                termNumber:    cells[0] ? cells[0].innerText.trim() : "",
                effectiveDate: cells[1] ? cells[1].innerText.trim() : "",
                expirationDate:cells[2] ? cells[2].innerText.trim() : "",
                premium:       cells[3] ? cells[3].innerText.trim() : "",
                status:        cells[4] ? cells[4].innerText.trim() : "",
                txnType:       cells[5] ? cells[5].innerText.trim() : ""
            });
        }
    });
    return JSON.stringify(terms);
''' Result=> TermsJSON
