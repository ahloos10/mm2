# PowerApps — All Screen Formulas
# Each screen follows the same layout pattern:
#
#   cmpHeader           (Header Component — top 64px)
#   rectBackground      (fills remaining screen, Secondary color)
#   rectCard            (white card, centered, holds all inputs)
#   [input controls]
#   btnSubmit
#   lblStatus
#   spinnerOverlay      (optional — visible when varIsLoading)
#
# Standard card dimensions (tablet 1366x768):
#   Card: X:32  Y:80  Width:Parent.Width-64  Height:Parent.Height-96

# ───────────────────────────────────────────────────────────────────────────
# SCREEN: CreateNoteScreen
# ───────────────────────────────────────────────────────────────────────────
# cmpHeader.ScreenTitle = "Create Note"
# cmpHeader.PolicyNumber = varActivePolicyNumber

# txtNoteSubject
# HintText: "Subject"   X:48  Y:112  Width:600  Height:40

# ddlNoteTopic
# Items: ["General","Underwriting","Billing","Claims","Compliance","Other"]
# X:48  Y:164  Width:280  Height:40

# txtNoteBody
# HintText: "Note text..."  Mode: Multiline
# X:48  Y:216  Width:800  Height:120

# btnSubmit OnSelect:
If(IsBlank(txtNoteSubject.Text) || IsBlank(txtNoteBody.Text),
    Notify("Subject and Note are required.", NotificationType.Warning),
    Set(varIsLoading,true);
    Set(varResult, PC_CreatePolicyNote.Run(varActivePolicyNumber, txtNoteSubject.Text, txtNoteBody.Text, ddlNoteTopic.Selected.Value));
    Set(varIsLoading,false);
    Notify(varResult.Result, If(StartsWith(varResult.Result,"Success"), NotificationType.Success, NotificationType.Error))
);


# ───────────────────────────────────────────────────────────────────────────
# SCREEN: ScreenScrapeScreen
# ───────────────────────────────────────────────────────────────────────────
# cmpHeader.ScreenTitle = "Screen Scrape"

# btnRunScrape OnSelect:
Set(varIsLoading,true);
Set(varScrapeResult, PC_ScreenScrape.Run(varActivePolicyNumber));
Set(varScrapeData,
    ForAll(Table(ParseJSON(varScrapeResult.ScrapedDataJSON)),
        {Tab: Text(ThisRecord.tab), Field: Text(ThisRecord.field), Value: Text(ThisRecord.value)}
    )
);
Set(varIsLoading,false);
Notify(CountRows(varScrapeData) & " fields captured.", NotificationType.Success);

# galScrapeResults Items: varScrapeData
# lblTab:   ThisItem.Tab    lblField: ThisItem.Field    lblValue: ThisItem.Value

# ddlFilterTab (filter gallery by tab)
# Items: Distinct(varScrapeData, Tab)
# galScrapeResults Items (filtered):
Filter(varScrapeData, Tab = ddlFilterTab.Selected.Value || ddlFilterTab.Selected.Value = "All")

# ───────────────────────────────────────────────────────────────────────────
# SCREEN: CancelPolicyScreen
# ───────────────────────────────────────────────────────────────────────────
# cmpHeader.ScreenTitle = "Cancel Policy"

# ddlSource   Items: ["Insured","Agency","Carrier"]
# ddlReason   Items: ["Insured Request","Non-Payment","Underwriting","Fraud","Other"]
# dateEffective
# ddlRefund   Items: ["Pro Rata","Short Rate","Flat"]

# rectDangerBanner
# Fill: RGBA(255,235,233,1)  BorderColor: First(colTheme).Danger  BorderThickness:1
# Y: above submit button

# lblDangerWarning
# Text: "⚠ IRREVERSIBLE — This will cancel the policy. Confirm all details before submitting."
# Color: First(colTheme).Danger  Size:12

# btnSubmit
# Fill: First(colTheme).Danger  Color:White
# Text: "Cancel Policy"
# OnSelect:
If(
    Not(varCancelConfirmed),
    UpdateContext({varCancelConfirmed: true});
    Notify("Tap Cancel Policy again to confirm.", NotificationType.Warning),
    // second tap — confirmed
    Set(varIsLoading,true);
    Set(varResult, PC_CancelPolicy.Run(varActivePolicyNumber, ddlReason.Selected.Value, ddlSource.Selected.Value, Text(dateEffective.SelectedDate,"mm/dd/yyyy"), ddlRefund.Selected.Value));
    Set(varIsLoading,false);
    UpdateContext({varCancelConfirmed: false});
    Notify(varResult.Result, If(StartsWith(varResult.Result,"Success"), NotificationType.Success, NotificationType.Error))
);

# btnSubmit Text:
If(varCancelConfirmed, "⚠ Confirm Cancel", "Cancel Policy")


# ───────────────────────────────────────────────────────────────────────────
# SCREEN: PolicyTermsScreen
# ───────────────────────────────────────────────────────────────────────────
# cmpHeader.ScreenTitle = "Policy Terms"

# btnLoadTerms OnSelect:
Set(varIsLoading,true);
Set(varTermsResult, PC_IteratePolicyTerms.Run(varActivePolicyNumber));
Set(varTermsData,
    ForAll(Table(ParseJSON(varTermsResult.TermsJSON)),
        {TermNumber: Text(ThisRecord.termNumber), EffDate: Text(ThisRecord.effectiveDate),
         ExpDate: Text(ThisRecord.expirationDate), Premium: Text(ThisRecord.premium),
         Status: Text(ThisRecord.status), TxnType: Text(ThisRecord.txnType)}
    )
);
Set(varIsLoading,false);
Notify(CountRows(varTermsData) & " terms loaded.", NotificationType.Success);

# galTerms Items: varTermsData
# Columns: Term #, Eff Date, Exp Date, Premium, Status, Txn Type
# lblTermStatus Color:
If(ThisItem.Status="Active", First(colTheme).Success,
   ThisItem.Status="Cancelled", First(colTheme).Danger,
   First(colTheme).TextSecondary)


# ───────────────────────────────────────────────────────────────────────────
# SCREEN: BulkProducerScreen
# ───────────────────────────────────────────────────────────────────────────
# cmpHeader.ScreenTitle = "Bulk Producer Change"

# txtPolicyList    HintText: "One policy number per line"  Multiline  Height:160
# txtNewProducer   HintText: "New Producer Name (must match PolicyCenter exactly)"
# txtProducerCode  HintText: "Producer Code"
# dateEffective

# lblPolicyCount
# Text: CountRows(Split(txtPolicyList.Text, Char(10))) & " policies entered"
# Color: First(colTheme).TextSecondary  Size:11

# btnRun OnSelect:
Set(varPolicyArray,
    Concat(ForAll(Split(txtPolicyList.Text,Char(10)),
        If(Not(IsBlank(Trim(Result))), """" & Trim(Result) & """,", "")
    ), Value)
);
Set(varPolicyJSON, "[" & Left(varPolicyArray,Len(varPolicyArray)-1) & "]");
Set(varIsLoading,true);
Set(varBulkResult, PC_BulkProducerChange.Run(varPolicyJSON, txtNewProducer.Text, txtProducerCode.Text, Text(dateEffective.SelectedDate,"mm/dd/yyyy")));
Set(varBulkData,
    ForAll(Table(ParseJSON(varBulkResult.ResultsJSON)),
        {Policy: Text(ThisRecord.policy), Status: Text(ThisRecord.status), Message: Text(ThisRecord.message)}
    )
);
Set(varIsLoading,false);
Notify(
    CountRows(Filter(varBulkData,Status="Success")) & " succeeded, " & CountRows(Filter(varBulkData,Status<>"Success")) & " failed.",
    If(CountRows(Filter(varBulkData,Status<>"Success"))=0, NotificationType.Success, NotificationType.Warning)
);

# galBulkResults Items: varBulkData
# lblPolicy: ThisItem.Policy
# lblStatus Color: If(ThisItem.Status="Success", First(colTheme).Success, First(colTheme).Danger)
# lblMessage: If(ThisItem.Status<>"Success", ThisItem.Message, "")
