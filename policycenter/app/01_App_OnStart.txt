# PowerApps — App.OnStart
# Set this in the App object > OnStart property.
# Runs once when the app launches. Initializes theme, globals, and navigation state.

ClearCollect(
    colTheme,
    {
        Primary:        RGBA(0,  70, 127, 1),   // deep blue (header/buttons)
        PrimaryLight:   RGBA(0, 120, 212, 1),   // lighter blue (hover/accent)
        Secondary:      RGBA(243, 242, 241, 1), // light grey (screen background)
        Surface:        RGBA(255,255,255, 1),   // white (card background)
        Danger:         RGBA(196, 43, 28, 1),   // red (cancel/error)
        Success:        RGBA(16, 124, 16, 1),   // green (success messages)
        TextPrimary:    RGBA(32, 31, 30, 1),    // near-black (body text)
        TextSecondary:  RGBA(96, 94, 92, 1),    // grey (labels/secondary)
        Border:         RGBA(200,198,196, 1)    // light border
    }
);

// Active policy context — shared across all screens
Set(varActivePolicyNumber, "");
Set(varActivePolicyLoaded, false);

// Navigation state
Set(varCurrentScreen, "Home");

// App version
Set(varAppVersion, "1.0");

// Loading state
Set(varIsLoading, false);
Set(varStatusMessage, "");

// Navigate to home
Navigate(HomeScreen, ScreenTransition.None);
