    IsBlank(txtPolicySearch.Text),
    Notify("Please enter a policy number.", NotificationType.Warning),
    Set(varActivePolicyNumber, Trim(txtPolicySearch.Text));
    Set(varActivePolicyLoaded, true);
    Notify("Policy set: " & varActivePolicyNumber, NotificationType.Success)
);
